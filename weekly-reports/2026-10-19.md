  document.addEventListener('keydown', (e) => { if (e.key === 'Escape' && ov.style.display !== 'none') { e.stopPropagation(); close(); } }, true);
  const open = (src) => { im.src = src; ov.style.display = 'flex'; };
  const ready = () => {
    document.body.appendChild(ov);
    document.addEventListener('click', (e) => {
      const t = e.target;
      if (t && t.tagName === 'IMG' && t.closest('section') && !t.classList.contains('emoji')) { e.preventDefault(); e.stopPropagation(); open(t.currentSrc || t.src); }
    }, true);
  };
  document.readyState === 'loading' ? document.addEventListener('DOMContentLoaded', ready) : ready();
})();
</script>
